# ─── محافظت با رمز عبور (Basic Auth) ─────────────────────────────────
# پیش از عمومی‌کردن سایت، این بلوک را به فایل .htaccess ریشه پروژه اضافه کنید.
# مسیر فایل رمزها را با مسیر واقعی روی سرور جایگزین کنید (بیرون از public_html بهتر است).

AuthType Basic
AuthName "سرمایه‌یار — دسترسی محدود"
AuthUserFile /home/USERNAME/.htpasswd-sarmayeyar
Require valid-user

# ساخت فایل رمز (روی سرور، در خط فرمان):
#   htpasswd -c /home/USERNAME/.htpasswd-sarmayeyar myuser
# اگر htpasswd ندارید، همین کار را با پایتون انجام دهید:
#   python3 -c "import crypt;print('myuser:'+crypt.crypt('MyPass123','sa'))" > /home/USERNAME/.htpasswd-sarmayeyar
